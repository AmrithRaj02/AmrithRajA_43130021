package basic4;
import java.util.ArrayList;
import java.util.Iterator;
import java.util.LinkedList;
public class ArrayListLinkedList{
	public static void main(String[] args) {
		ArrayList<String> list = new ArrayList<String>();
		list.add("Harjith");
		list.add("Sanjay");
		System.out.println("ArrayList : ");
		System.out.println(list);
		Iterator it = list.iterator();
		while (it.hasNext()) {
			System.out.println(it.next());
		}
		System.out.println("---------------------");	
		LinkedList<String> ll = new LinkedList<String>();
		ll.add("Jeff");
		ll.add("Arjun");
		System.out.println("LinkedList : ");
		System.out.println(ll);
		Iterator <String> itr = ll.iterator();
		while(itr.hasNext()) {
			System.out.println(itr.next());
		}
	}
}

package basic4;
import java.util.*;  
public class LinkedListAdd{  
 public static void main(String args[]){  
 LinkedList<String> ll=new LinkedList<String>();  
           System.out.println("Initial elements: "+ll);  
           ll.add("Slavia");  
           ll.add("Virtus");  
           ll.add("Verna");  
           System.out.println("After adding the first time: "+ll);              
           ll.add(1, "Crysta");  
           System.out.println("after adding new name in middle: "+ll);             
           LinkedList<String> ll2=new LinkedList<String>();  
           ll2.add("Creta");  
           ll2.add("Breeza");               
           ll.addAll(ll2);  
           System.out.println("joining both linkedlists: "+ll);  
           LinkedList<String> ll3=new LinkedList<String>();  
           ll3.add("Force");  
           ll3.add("Maxicab");  
           ll.addAll(1, ll3);  
           System.out.println("joining truck in middle: "+ll);           
           ll.addFirst("Ajay");  
           System.out.println("add name in first place: "+ll);             
           ll.addLast("Amrith");  
           System.out.println("add name in last space: "+ll);              
 }  
}

package basic4;
import java.util.*;  
public class LinkedListRemove {  
        public static void main(String [] args)  
        {  
           LinkedList<String> l=new LinkedList<String>();  
           l.add("car1");  
           l.add("car2");  
           l.add("car3");  
           l.add("car4");  
           l.add("car5");  
           l.add("car6");  
           l.add("car7");  
           l.add("car8");  
           l.add("car9(1)");
           l.add("car9(1)");
           l.add("car9(2)");  
           System.out.println("Initial list of elements: "+l);          
              l.remove("car4");  
              System.out.println("After removing car4: "+l);   
              l.remove(0);  
              System.out.println("after removing first element (using index value): "+l);   
              LinkedList<String> l2=new LinkedList<String>();  
              l2.add("truck1");  
              l2.add("truck2");  
              l.addAll(l2);  
              System.out.println("adding both linked list : "+l);   
              l.removeAll(l2);  
              System.out.println("removing elements of second linkedlist: "+l);   
              l.removeFirst();  
              System.out.println("after removing first element: "+l);  
              l.removeLast();  
              System.out.println("After removing last element: "+l);             
              l.removeFirstOccurrence("car9(1)");  
              System.out.println("removing the first of duplicates: "+l);          
              l.removeLastOccurrence("car9(1)");  
              System.out.println("removing last of duplicates: "+l);                
              l.clear();  
              System.out.println("clearing all: "+l);   
       }
        
}
